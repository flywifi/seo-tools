# Optional media backends for the videoedit tools (P29). NOTHING here is required:
# every tool in tools/videoedit/ runs stdlib-only and degrades to the transcript floor
# (shared/docintel/transcripts.py) when a backend is absent. Install only what you want;
# preflight (python3 tools/videoedit/preflight.py) reports what was found.
#
# ffmpeg and melt are system binaries, not pip packages: install via your OS package
# manager (brew install ffmpeg mlt / apt install ffmpeg melt) or a static build.

# Scene/chapter detection (recommended detector; catches isoluminant cuts ffmpeg scdet misses)
scenedetect>=0.7,<0.8
opencv-python-headless

# In-process silence detection without an ffmpeg binary (validated 18.1.0 under Python 3.12, P80)
av>=18,<19

# Frame-array math in mediaprobe (also pulled transitively by opencv/moviepy).
# numpy 2.5 dropped Python 3.11, which is why the repo floor is 3.12 (tools/setup.py). Validated
# 2.5.2 under Python 3.12.3 against the P26 goldens (docs/video-tooling-integration-evidence.json,
# revalidations). The <3 bound is the numpy 2.x ABI the opencv/moviepy wheels are built against.
numpy>=2.5,<3

# Local Shorts reframe rendering (crop geometry itself needs nothing)
moviepy>=2,<3

# Timeline interchange extras (adapters are separate plugins since OTIO 0.18)
opentimelineio
